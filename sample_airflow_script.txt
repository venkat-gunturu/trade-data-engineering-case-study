from datetime import datetime, timedelta
from airflow import DAG
from airflow.operators.bash import BashOperator
from airflow.utils.email import send_email

# 1. Define the success email callback function
def send_dag_success_email(context):
    """Sends an email notification when the entire DAG succeeds."""
    dag_id = context['dag'].dag_id
    execution_date = context['data_interval_end'] # Use 'execution_date' in older Airflow versions
    
    subject = f"Airflow Alert: DAG [{dag_id}] Succeeded!"
    body = f"""
    <h3>DAG Execution Success</h3>
    <p><b>DAG ID:</b> {dag_id}</p>
    <p><b>Execution Date:</b> {execution_date}</p>
    <p>The workflow completed successfully without any unhandled errors.</p>
    """
    
    # Airflow looks for configured SMTP credentials automatically
    send_email(
        to=['team-alerts@example.com'], 
        subject=subject, 
        html_content=body
    )

# 2. Set default arguments for Task-Level settings
default_args = {
    'owner': 'data_ops',
    'start_date': datetime(2026, 1, 1),
    'retries': 1,
    'retry_delay': timedelta(minutes=5),
    # Built-in email parameters (Triggers per-task failure)
    'email': ['team-alerts@example.com'],
    'email_on_failure': True,
    'email_on_retry': False,
}

# 3. Instantiate the DAG with the success callback
with DAG(
    dag_id='pipeline_with_notifications',
    default_args=default_args,
    schedule_interval='@daily',
    catchup=False,
    on_success_callback=send_dag_success_email, # Triggered if all tasks succeed
) as dag:

    task_1 = BashOperator(
        task_id='extract_data',
        bash_command='echo "Extracting data..."',
    )

    task_2 = BashOperator(
        task_id='transform_data',
        bash_command='echo "Transforming data..."',
    )

    task_1 >> task_2
